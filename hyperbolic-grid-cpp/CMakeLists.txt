cmake_minimum_required(VERSION 3.16)
project(hypergrid LANGUAGES CXX)
add_executable(hypergrid src/main.cpp)
target_compile_features(hypergrid PRIVATE cxx_std_17)
set_target_properties(hypergrid PROPERTIES CXX_EXTENSIONS OFF)
if(MSVC)
    target_compile_options(hypergrid PRIVATE /W4 /permissive- /utf-8)
else()
    target_compile_options(hypergrid PRIVATE -Wall -Wextra -Wpedantic)
endif()
include(CTest)
if(BUILD_TESTING)
    find_package(Python3 COMPONENTS Interpreter QUIET)
    if(Python3_Interpreter_FOUND)
        add_test(NAME hypergrid_regression COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/regression.py" "$<TARGET_FILE:hypergrid>" "${CMAKE_CURRENT_SOURCE_DIR}/examples")
    endif()
endif()
